function setBreakStatus(type) { let state = getUserState(); // Block taking breaks if not currently working if (!state.isWorking && type !== 'Working') { alert("Please start work first before taking a break!"); return; } // Block double-tapping the exact same break or switching breaks without clicking "Back to Seat" if (state.status === type && type !== 'Working') { alert(`You are already on a ${type}!`); return; } if (state.status.includes('Break') && type !== 'Working') { alert("You are currently in a break! Click 'Back to Seat' to finish your break first."); return; } if (type === 'Working') { // Resume Work (Back to Seat) if (state.lastBreakStartTimestamp) { let breakElapsed = Math.floor((Date.now() - state.lastBreakStartTimestamp) / 1000); state.totalBreakSec = (state.totalBreakSec || 0) + breakElapsed; state.lastBreakStartTimestamp = null; } state.status = 'Working'; state.breakEnd = null; state.lastStartTimestamp = Date.now(); } else { // Pause shift timer & record break start if (state.lastStartTimestamp) { let elapsed = Math.floor((Date.now() - state.lastStartTimestamp) / 1000); state.accumulatedSec = (state.accumulatedSec || 0) + elapsed; state.lastStartTimestamp = null; } state.status = type; state.lastBreakStartTimestamp = Date.now(); if (type === 'Smoke / Toilet Break') { state.breakEnd = Date.now() + (15 * 60 * 1000); } else if (type === 'Eat Break') { state.breakEnd = Date.now() + (60 * 60 * 1000); } } saveAndSyncState(state); }
